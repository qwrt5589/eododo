-- DataStore保存收藏
local success, dataStore = pcall(function()
    return game:GetService("DataStoreService"):GetDataStore("UI_Favorites_v4")
end)

local function LoadFavorites()
    if success then
        local playerId = tostring(game.Players.LocalPlayer.UserId)
        local success2, saved = pcall(function()
            return dataStore:GetAsync(playerId)
        end)
        if success2 and saved then
            library.favorites = saved
        end
    end
end

local function SaveFavorites()
    if success then
        local playerId = tostring(game.Players.LocalPlayer.UserId)
        pcall(function()
            dataStore:SetAsync(playerId, library.favorites)
        end)
    end
end

LoadFavorites()

-- 收藏管理函数
function library:AddFavorite(buttonName, callback, sectionName, tabName)
    if not library.favorites[buttonName] then
        library.favorites[buttonName] = {
            timestamp = os.time(),
            sectionName = sectionName,
            tabName = tabName
        }
        library.favoriteCallbacks[buttonName] = callback
        library.favoriteUpdateSignal:Fire()
        SaveFavorites()
        return true
    end
    return false
end

function library:RemoveFavorite(buttonName)
    if library.favorites[buttonName] then
        library.favorites[buttonName] = nil
        library.favoriteCallbacks[buttonName] = nil
        library.favoriteUpdateSignal:Fire()
        SaveFavorites()
        return true
    end
    return false
end

function library:ToggleFavorite(buttonName, callback, sectionName, tabName)
    if library.favorites[buttonName] then
        return library:RemoveFavorite(buttonName)
    else
        return library:AddFavorite(buttonName, callback, sectionName, tabName)
    end
end

function library:IsFavorite(buttonName)
    return library.favorites[buttonName] ~= nil
end

function library:GetFavoriteCallback(buttonName)
    return library.favoriteCallbacks[buttonName]
end

function GetSortedFavorites()
    local sorted = {}
    for name, data in pairs(library.favorites) do
        table.insert(sorted, {
            name = name,
            timestamp = data.timestamp,
            sectionName = data.sectionName,
            tabName = data.tabName
        })
    end
    table.sort(sorted, function(a, b)
        return a.timestamp > b.timestamp
    end)
    return sorted
end

function GetRainbowColor()
    return RainbowColors[currentColorIndex]
end

function AnimateTextColor(textLabel)
    spawn(function()
        while textLabel and textLabel.Parent do
            local nextIndex = currentColorIndex % #RainbowColors + 1
            local nextColor = RainbowColors[nextIndex]
            
            services.TweenService:Create(textLabel, TweenInfo.new(colorTransitionSpeed, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), {
                TextColor3 = nextColor
            }):Play()
            task.wait(colorTransitionSpeed)
            
            if not textLabel or not textLabel.Parent then break end
        end
    end)
end

function Tween(obj, t, data)
	services.TweenService
		:Create(obj, TweenInfo.new(t[1], Enum.EasingStyle[t[2]], Enum.EasingDirection[t[3]]), data)
		:Play()
	return true
end

-- 按钮动画控制器
local ButtonAnimations = {}
local ANIMATION_DEBOUNCE = 0.25

-- 高级果冻效果
function SafeJellyEffect(button, buttonName)
    if not button or not button.Parent then return end
    
    if ButtonAnimations[buttonName] and (os.clock() - ButtonAnimations[buttonName]) < ANIMATION_DEBOUNCE then
        return
    end
    
    ButtonAnimations[buttonName] = os.clock()
    
    spawn(function()
        local originalSize = button.Size
        
        local tween1 = services.TweenService:Create(button, TweenInfo.new(0.06, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.new(originalSize.X.Scale * 0.94, 0, originalSize.Y.Scale * 0.94, 0)
        })
        tween1:Play()
        
        task.wait(0.06)
        
        if not button or not button.Parent then return end
        
        local tween2 = services.TweenService:Create(button, TweenInfo.new(0.1, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out, 0, false, 0), {
            Size = originalSize
        })
        tween2:Play()
    end)
end

        return

        local tween2 = services.TweenService:Create(button, TweenInfo.new(0.12, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out, 0, false, 0), {

-- Tab切换系统
local toggled = false
local switchingTabs = false

function switchTab(new)
	if switchingTabs then
		return
	end
	local old = library.currentTab
	if old == nil then
		new[2].Visible = true
		library.currentTab = new
		services.TweenService:Create(new[1], TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { ImageTransparency = 0 }):Play()
		services.TweenService:Create(new[1].TabText, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { TextTransparency = 0 }):Play()
		return
	end
	if old[1] == new[1] then
		return
	end
	switchingTabs = true
	library.currentTab = new
	services.TweenService:Create(old[1], TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), { ImageTransparency = 0.3 }):Play()
	services.TweenService:Create(new[1], TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), { ImageTransparency = 0 }):Play()
	services.TweenService:Create(old[1].TabText, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), { TextTransparency = 0.3 }):Play()
	services.TweenService:Create(new[1].TabText, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), { TextTransparency = 0 }):Play()
	old[2].Visible = false
	new[2].Visible = true
	task.wait(0.12)
	switchingTabs = false
end

-- 窗口拖动系统
function drag(frame, hold)
	if not hold then
		hold = frame
	end
	local dragging
	local dragInput
	local dragStart
	local startPos
	local function update(input)
		local delta = input.Position - dragStart
		frame.Position =
			UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
	hold.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)
	frame.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement then
			dragInput = input
		end
	end)
	services.UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			update(input)
		end
	end)
end

						for _, module in pairs(Objs:GetChildren()) do
							if module:IsA("Frame") and module.Name == "BtnModule" then
								local btn = module:FindFirstChild("Btn")
								if btn then

										BackgroundColor3 = zyColor,
										BackgroundTransparency = zyColorTransparency,

									}):Play()
								end
							end
						end
						return 
					end
					
					ClearButton.Visible = true
					local foundCount = 0
					local searchLower = searchText:lower()
					
					for _, module in pairs(Objs:GetChildren()) do
						if module:IsA("Frame") and module.Name == "BtnModule" then
							local btn = module:FindFirstChild("Btn")
							if btn and module ~= TextboxModule then
								module.LayoutOrder = originalLayoutOrders[btn.Text] or module.LayoutOrder
								services.TweenService:Create(btn, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {

								}):Play()
							end
						end
					end
					
					for _, module in pairs(Objs:GetChildren()) do
						if module:IsA("Frame") and module.Name == "BtnModule" then
							local btn = module:FindFirstChild("Btn")
							if btn and module ~= TextboxModule then
								local btnText = btn.Text:gsub("^%s+", ""):gsub("%s+$", "")
								
								if btnText:lower():find(searchLower, 1, true) then

									module.LayoutOrder = -1000 + foundCount
									
									local buttonName = "SearchResult_" .. btnText
									SafeJellyEffect(btn, buttonName)
									
									services.TweenService:Create(btn, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {

											}):Play()

							Position = UDim2.new(0.25, 5, 0.26, 0)

						task.wait(0.05)
